## WXML 渲染建议

```xml
<block wx:for="{{prizeTables}}" wx:key="type">
  <view class="prize-block prize-{{item.theme}}">
    <view class="prize-title">{{item.title}}</view>
    <view wx:if="{{item.groups}}" class="group-head-row">
      <view wx:for="{{item.groups}}" wx:key="*this" class="group-head">{{item}}</view>
    </view>
    <view wx:else class="stage-head-row">
      <view wx:for="{{item.headers}}" wx:key="*this" class="stage-head">{{item}}</view>
    </view>
    <!-- groups 表：每组是 [名次,金额]，stage 表：每列由 columns[col][row] 取值。 -->
    <view wx:if="{{item.groups}}" wx:for="{{item.rows}}" wx:for-item="row" wx:key="index" class="prize-row">...</view>
    <view wx:else wx:for="{{item.rows}}" wx:for-item="row" wx:key="index" class="prize-row">...</view>
    <view class="prize-total-row">
      <view wx:for="{{item.subtotals}}" wx:key="*this" class="prize-total">{{item}}</view>
    </view>
  </view>
</block>
```
